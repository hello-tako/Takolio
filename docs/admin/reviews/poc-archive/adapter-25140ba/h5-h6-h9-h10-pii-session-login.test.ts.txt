import { describe, expect, it } from 'vitest';
import { DEV_BASE, FX, fixtureDb, login, makeEnv, read, rows, send } from './harness';

const t = fixtureDb();

describe('H5: PII reveal and audit', () => {
  it('H5 CONFIRMED: email is returned even though the audit insert fails', async () => {
    const env = makeEnv(t.db);
    const s = await login(env, FX('super'));
    t.db.seed("CREATE TRIGGER poc_audit_down BEFORE INSERT ON admin_audit_events BEGIN SELECT RAISE(ABORT, 'audit store unavailable'); END;");
    const r = await read(env, s.session, '/api/v1/admin/users/fx_user_001?reveal=email');
    expect(r.status).toBe(200);
    expect(r.json.data.email).toBe('customer001@fixture.takosan.test');
    expect(rows(t.db, "SELECT COUNT(*) AS n FROM admin_audit_events WHERE action = 'users.pii.reveal'")[0].n).toBe(0);
  });

  it('control: with audit available the reveal is recorded', async () => {
    const env = makeEnv(t.db);
    const s = await login(env, FX('super'));
    const r = await read(env, s.session, '/api/v1/admin/users/fx_user_001?reveal=email');
    expect(r.json.data.email).toBe('customer001@fixture.takosan.test');
    expect(rows(t.db, "SELECT COUNT(*) AS n FROM admin_audit_events WHERE action = 'users.pii.reveal'")[0].n).toBe(1);
  });
});

describe('H6: credential change vs admin sessions', () => {
  it('H6 CONFIRMED: rotating the customer credential row leaves the admin session valid', async () => {
    const env = makeEnv(t.db);
    const s = await login(env, FX('super'));
    // Same row effect as the customer reset path (services/auth-otp.ts writes auth_accounts.password_hash/salt only).
    t.db.seed(`UPDATE auth_accounts SET password_hash = '${'0'.repeat(64)}', salt = 'poc-rotated' WHERE user_id = 'fx_admin_super';`);
    expect((await read(env, s.session, '/api/v1/admin/me')).status).toBe(200);
    expect(rows(t.db, "SELECT revoked_at FROM admin_sessions WHERE user_id = 'fx_admin_super'")[0].revoked_at).toBeNull();
  });
});

describe('H9: login brute-force controls', () => {
  const attempt = (env: ReturnType<typeof makeEnv>, headers: Record<string, string>, cf = true) =>
    send(env, `${DEV_BASE}/api/v1/admin/auth/login`, { method: 'POST', cf, headers: { Origin: DEV_BASE, ...headers }, body: { email: FX('super'), password: 'wrong-password' } });

  it('H9a CONFIRMED: no per-account throttle; 10 wrong passwords from 10 IPs are all 401', async () => {
    const env = makeEnv(t.db);
    const statuses: number[] = [];
    for (let i = 1; i <= 10; i += 1) statuses.push((await attempt(env, { 'CF-Connecting-IP': `203.0.113.${i}` })).status);
    expect(statuses).toEqual(Array(10).fill(401));
  });

  it('H9b control: the per-IP limit (5/min) applies to one source', async () => {
    const env = makeEnv(t.db);
    const statuses: number[] = [];
    for (let i = 0; i < 6; i += 1) statuses.push((await attempt(env, { 'CF-Connecting-IP': '203.0.113.99' })).status);
    expect(statuses.slice(0, 5)).toEqual([401, 401, 401, 401, 401]);
    expect(statuses[5]).toBe(429);
  });

  it('H9c CONFIRMED (no CF header, e.g. non-Cloudflare ingress): rotating X-Forwarded-For bypasses the limiter', async () => {
    const env = makeEnv(t.db);
    const statuses: number[] = [];
    for (let i = 1; i <= 8; i += 1) statuses.push((await attempt(env, { 'X-Forwarded-For': `198.51.100.${i}` }, false)).status);
    expect(statuses).toEqual(Array(8).fill(401));
  });
});

describe('H10: data exposure to non-PII roles', () => {
  it('H10 CONFIRMED: users.read roles confirm a full email via search with no audit trail', async () => {
    const env = makeEnv(t.db);
    const aud = await login(env, FX('auditor'));
    const usersAuditBefore = rows(t.db, "SELECT COUNT(*) AS n FROM admin_audit_events WHERE actor_user_id = 'fx_admin_auditor' AND action LIKE 'users.%'")[0].n;
    const hit = await read(env, aud.session, `/api/v1/admin/users?q=${encodeURIComponent(FX('customer001'))}`);
    expect(hit.status).toBe(200);
    expect(hit.json.data.total).toBe(1);
    expect(hit.json.data.items[0].id).toBe('fx_user_001');
    expect(hit.text).not.toContain('customer001@fixture');
    const miss = await read(env, aud.session, `/api/v1/admin/users?q=${encodeURIComponent(FX('no-such-person'))}`);
    expect(miss.json.data.total).toBe(0);
    expect(rows(t.db, "SELECT COUNT(*) AS n FROM admin_audit_events WHERE actor_user_id = 'fx_admin_auditor' AND action LIKE 'users.%'")[0].n).toBe(usersAuditBefore);
  });

  it('control: the same auditor is denied the explicit reveal (403) and that denial is audited', async () => {
    const env = makeEnv(t.db);
    const aud = await login(env, FX('auditor'));
    const r = await read(env, aud.session, '/api/v1/admin/users/fx_user_001?reveal=email');
    expect(r.status).toBe(403);
    expect(rows(t.db, "SELECT COUNT(*) AS n FROM admin_audit_events WHERE action = 'admin.permission.denied' AND actor_user_id = 'fx_admin_auditor'")[0].n).toBe(1);
  });
});
